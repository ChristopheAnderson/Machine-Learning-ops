PRAGMA cipher_page_size = 4096;
PRAGMA kdf_iter = 600000;
PRAGMA journal_mode = WAL;
PRAGMA synchronous = FULL;

-- 2. Sauvegarde atomique à chaud sans bloquer les lectures/écritures
VACUUM INTO '/srv/backups/airgap/snapshot_hot_encrypted.db';

Commande Pipeline Audio Hors Ligne (FFmpeg + Whisper CTranslate2 INT8 + Pyannote)

                        

                            
                            
# 1. Normalisation acoustique 16 kHz Mono PCM + Filtre passe-bande vocal (80Hz - 7600Hz)
ffmpeg -i input_audio.wav -af "highpass=f=80,lowpass=f=7600,dynaudnorm" -ar 16000 -ac 1 norm_16k.wav

# 2. Exécution conjointe Diarisation (Pyannote 3.1) + Transcription (Faster-Whisper Large-v3 INT8)
python -m src.module4_audio_asr_diarization_qa --input norm_16k.wav --device cuda \
  --compute-type int8_float16 --vad-filter silero_v5 --diarization-clustering spectral
